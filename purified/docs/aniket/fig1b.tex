\documentclass[tikz,border=8pt]{standalone}
\usepackage{amsmath,amssymb}
\usetikzlibrary{arrows.meta,positioning,calc}

\begin{document}
\begin{tikzpicture}[
    >=Latex,
    font=\Large,
    line width=1pt,
    every node/.style={align=center},
    act/.style={
        draw,
        rounded corners=8pt,
        minimum width=2.3cm,
        minimum height=1.45cm,
        fill=white
    },
    weight/.style={
        draw,
        rounded corners=8pt,
        minimum width=2.6cm,
        minimum height=1.45cm,
        fill=white
    },
    inA/.style={draw, rounded corners=8pt, minimum width=2.3cm, minimum height=1.45cm, fill=violet!15},
    inB/.style={draw, rounded corners=8pt, minimum width=2.3cm, minimum height=1.45cm, fill=blue!12},
    inC/.style={draw, rounded corners=8pt, minimum width=2.3cm, minimum height=1.45cm, fill=yellow!20},
    inD/.style={draw, rounded corners=8pt, minimum width=2.3cm, minimum height=1.45cm, fill=red!12}
]

% =========================================================
% Geometry constants — keep the diagram symmetric around
% the latent column's vertical center (y = -1.0).
% =========================================================
% Row y positions (5 rows: 4 token boxes + 1 vdots), evenly
% spaced so the centroid lands exactly at y = -1.0.
\def\yA{3.6}     % x_t / x_hat_t
\def\yB{1.3}     % x_{t+1}
\def\yC{-1.0}    % x_{t+2}
\def\yvd{-3.3}   % vdots row
\def\yD{-5.6}    % x_{t+T}
% x positions of the four columns (input, encoder, decoder, output)
\def\xIn{0}     \def\xEnc{3.8}    \def\xDec{13.2}   \def\xOut{17.0}
% Sum node (column between encoder and latent)
\def\xSum{7.0}  \def\ySum{-1.0}
% Latent column (anchored to topmost box top edge & bottommost box
% bottom edge so the green panel always frames the rows)
\def\xLatL{8.9} \def\xLatR{10.7}
\def\yLatTop{4.325}  \def\yLatBot{-6.325}
\def\xLatMid{9.8}
\def\yLatMid{-1.0}   % vertical center of the latent box

% Latent rows — 6 rows of equal height (u header + 4 z labels + vdots),
% evenly distributed across the latent column.
% row height = 10.65 / 6 = 1.775
\def\rowH{1.775}
% Divider y-coordinates (5 horizontal lines inside the box)
\def\yDivOne{2.55}      % below u  / above z_0
\def\yDivTwo{0.775}     % below z_0 / above z_1
\def\yDivThree{-1.0}    % below z_1 / above z_2
\def\yDivFour{-2.775}   % below z_2 / above vdots
\def\yDivFive{-4.55}    % below vdots / above z_{d-1}
% Row centers (midpoints of the regions defined by the dividers)
\def\yRowU{3.4375}      % u
\def\yRowZ0{1.6625}     % z_0
\def\yRowZ1{-0.1125}    % z_1
\def\yRowZ2{-1.8875}    % z_2
\def\yRowVd{-3.6625}    % vdots
\def\yRowZD{-5.4375}    % z_{d-1}

%------------------------------------------------
% Left column: inputs
%------------------------------------------------
\node[inA] (xt)   at (\xIn,  \yA)  {$x_t$};
\node[inB] (xt1)  at (\xIn,  \yB)  {$x_{t+1}$};
\node[inC] (xt2)  at (\xIn,  \yC)  {$x_{t+2}$};
\node      (vdL)  at (\xIn,  \yvd) {$\vdots$};
\node[inD] (xtT)  at (\xIn,  \yD)  {$x_{t+T}$};

% Encoder blocks
\node[weight] (Wenct)  at (\xEnc, \yA)  {$W^{t}_{\mathrm{enc}}$};
\node[weight] (Wenct1) at (\xEnc, \yB)  {$W^{t+1}_{\mathrm{enc}}$};
\node[weight] (Wenct2) at (\xEnc, \yC)  {$W^{t+2}_{\mathrm{enc}}$};
\node         (vdLe)   at (\xEnc, \yvd) {$\vdots$};
\node[weight] (WenctT) at (\xEnc, \yD)  {$W^{t+T}_{\mathrm{enc}}$};

% Arrows: inputs -> encoders
\draw[->] (xt)  -- (Wenct);
\draw[->] (xt1) -- (Wenct1);
\draw[->] (xt2) -- (Wenct2);
\draw[->] (xtT) -- (WenctT);

%------------------------------------------------
% Sum node — at the vertical center of the latent column
%------------------------------------------------
\node[draw, circle, minimum size=1.5cm, inner sep=0pt]
    (sum) at (\xSum, \ySum) {$\sum$};

% Arrows: encoders -> sum (TikZ picks the natural attachment point on
% the sum circle, so all four arrows aim at the sum centre and read
% symmetrically regardless of the encoder's row).
\draw[->] (Wenct.east)  -- (sum);
\draw[->] (Wenct1.east) -- (sum);
\draw[->] (Wenct2.east) -- (sum);
\draw[->] (WenctT.east) -- (sum);

%------------------------------------------------
% Shared latent vector — green panel, evenly partitioned
%------------------------------------------------
\node[font=\Large] (latenttitle) at (\xLatMid, 5.5) {shared latent};

\coordinate (latentNW) at (\xLatL, \yLatTop);
\coordinate (latentSE) at (\xLatR, \yLatBot);
\draw[rounded corners=10pt, fill=green!10]
    (latentNW) rectangle (latentSE);

% Horizontal dividers (evenly spaced — five lines partition the box
% into six equal-height rows: u, z_0, z_1, z_2, vdots, z_{d-1}).
\draw (\xLatL, \yDivOne)   -- (\xLatR, \yDivOne);
\draw (\xLatL, \yDivTwo)   -- (\xLatR, \yDivTwo);
\draw (\xLatL, \yDivThree) -- (\xLatR, \yDivThree);
\draw (\xLatL, \yDivFour)  -- (\xLatR, \yDivFour);
\draw (\xLatL, \yDivFive)  -- (\xLatR, \yDivFive);

% Row labels (vertically centered in their slot)
\node[font=\bfseries\Large] at (\xLatMid, \yRowU)  {$\mathbf{u}$};
\node                         at (\xLatMid, \yRowZ0) {$z_0$};
\node                         at (\xLatMid, \yRowZ1) {$z_1$};
\node                         at (\xLatMid, \yRowZ2) {$z_2$};
\node                         at (\xLatMid, \yRowVd) {$\vdots$};
\node                         at (\xLatMid, \yRowZD) {$z_{d-1}$};

% Sum -> latent (horizontal arrow at latent vertical center)
\draw[->] (sum.east) -- (\xLatL, \yLatMid);

%------------------------------------------------
% Decoder blocks
%------------------------------------------------
\node[weight] (Wdect)  at (\xDec, \yA)  {$W^{t}_{\mathrm{dec}}$};
\node[weight] (Wdect1) at (\xDec, \yB)  {$W^{t+1}_{\mathrm{dec}}$};
\node[weight] (Wdect2) at (\xDec, \yC)  {$W^{t+2}_{\mathrm{dec}}$};
\node         (vdRd)   at (\xDec, \yvd) {$\vdots$};
\node[weight] (WdectT) at (\xDec, \yD)  {$W^{t+T}_{\mathrm{dec}}$};

% Output blocks
\node[inA] (xhatT0) at (\xOut, \yA)  {$\hat{x}_t$};
\node[inB] (xhatT1) at (\xOut, \yB)  {$\hat{x}_{t+1}$};
\node[inC] (xhatT2) at (\xOut, \yC)  {$\hat{x}_{t+2}$};
\node      (vdR)    at (\xOut, \yvd) {$\vdots$};
\node[inD] (xhatTT) at (\xOut, \yD)  {$\hat{x}_{t+T}$};

% Arrows: latent -> decoders (originate at the latent box's right
% midpoint, mirroring the sum -> latent arrow on the left).
\coordinate (latentOut) at (\xLatR, \yLatMid);
\draw[->] (latentOut) -- (Wdect.west);
\draw[->] (latentOut) -- (Wdect1.west);
\draw[->] (latentOut) -- (Wdect2.west);
\draw[->] (latentOut) -- (WdectT.west);

% Arrows: decoders -> outputs
\draw[->] (Wdect)  -- (xhatT0);
\draw[->] (Wdect1) -- (xhatT1);
\draw[->] (Wdect2) -- (xhatT2);
\draw[->] (WdectT) -- (xhatTT);

\end{tikzpicture}
\end{document}
